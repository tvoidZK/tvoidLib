### Fabric code 
    import net.tvoid.lib.asset.ItemModels;

    public class YourModDataGenerator implements DataGeneratorEntrypoint {
        @Override
        public void onInitializeDataGenerator(FabricDataGenerator generator) {
            generator.createPack().addProvider(YourModModels::new);
        }

        static class YourModModels extends FabricModelProvider {
            YourModModels(FabricDataOutput out) { super(out); }

            @Override public void generateItemModels(ItemModelGenerators g) {
                ModItemModels.items(g, YourMod.ITEMS);
            }
        }
    }
